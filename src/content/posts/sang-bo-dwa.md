先看sub_401232

带入进去

sub_401232((unsigned __int8)(*(_BYTE *)(i + a1) + 7 * i), 3)

(a1 << a2) | (unsigned int)((int)a1 >> (8 - a2))

(((*(_BYTE *)(i + a1) + 7 * i)&0xff) << 3) | (unsigned int)(((*(_BYTE *)(i + a1) + 7 * i))&0xff >> (8 - 3))

`*(_BYTE *)(i + a1)`就是a1[i]

((((a1[i] + 7 * i)&0xff) << 3) | (unsigned int)(((a1[i] + 7 * i))&0xff >> (8 - 3))) & 0xff


v6[i] = byte_402049[i % 6] ^ ((((a1[i] + 7 * i)&0xff) << 3) | (unsigned int)(((a1[i] + 7 * i))&0xff >> (8 - 3))) & 0xff

(a1[i] + 7 * i) 左移3按位或右移7

反过来：

((((a1[i] + 7 * i)&0xff) << 3) | (unsigned int)(((a1[i] + 7 * i))&0xff >> (8 - 3))) & 0xff = v6[i] ^ byte_402049[i % 6]
